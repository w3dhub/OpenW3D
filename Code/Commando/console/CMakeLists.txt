cmake_minimum_required(VERSION 3.25)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OpenW3DServerConsole LANGUAGES CXX)
    include(CTest)
endif()

add_library(server_console STATIC ConsoleHost.cpp)
target_include_directories(server_console PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(server_console PUBLIC cxx_std_20)
if(WIN32)
    target_compile_definitions(server_console PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(server_console PUBLIC shell32)
endif()
if(MSVC)
    target_compile_options(server_console PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(server_console PRIVATE -Wall -Wextra -Werror)
endif()

if(BUILD_TESTING)
    add_executable(server_console_tests tests/ConsoleTests.cpp)
    target_link_libraries(server_console_tests PRIVATE server_console)
    add_test(NAME server_console_tests COMMAND server_console_tests)
endif()
